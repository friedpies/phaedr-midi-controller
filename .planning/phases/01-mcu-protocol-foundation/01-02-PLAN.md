**Step 0: Add LED_MAX_BRIGHTNESS to src/pinDefines.h (PWR-01)**

Append the following constant before the `#endif` in `src/pinDefines.h`. This caps SoftPWM brightness across all LEDs to keep total current draw within USB 500mA budget (24 LEDs × ~20mA peak × 70% duty ≈ 336mA avg + Teensy ~150mA = ~486mA; well within the 500mA USB limit).

```cpp
// USB power budget: 24 LEDs × ~20mA peak × (LED_MAX_BRIGHTNESS/255) duty + ~150mA Teensy <= 500mA
// 180/255 = 70.6% duty → ~14.1mA avg per LED → 24 × 14.1 + 150 = ~489mA max
#define LED_MAX_BRIGHTNESS 180
```

**src/mcuButton.h:**
```cpp
#ifndef mcu_button_h
#define mcu_button_h

#include <Arduino.h>
#include <Bounce2.h>
#include <SoftPWM.h>

class MCUButton
{
public:
    MCUButton(int buttonPin, int ledPin, int noteNum, int debounceTime);
    void init();
    void read();
    void setLedState(uint8_t velocity);  // 0=off, 127=on; 1=blink handled Phase 2
    int getLedPin() const;
    int getNoteNum() const;

private:
    int _buttonPin;
    int _ledPin;
    int _noteNum;
    int _debounceTime;
    bool _hasLed;  // false for buttons with no LED (Stop btn 15, Play btn 16)
    Bounce _bounce;
};

#endif
```

**src/mcuButton.cpp:**
```cpp
#include "mcuButton.h"

MCUButton::MCUButton(int buttonPin, int ledPin, int noteNum, int debounceTime)
    : _buttonPin(buttonPin), _ledPin(ledPin), _noteNum(noteNum),
      _debounceTime(debounceTime), _hasLed(ledPin >= 0)
{
}

void MCUButton::init()
{
    _bounce.attach(_buttonPin, INPUT_PULLUP);
    _bounce.interval(_debounceTime);
    if (_hasLed) {
        SoftPWMSet(_ledPin, 0);
        SoftPWMSetFadeTime(_ledPin, 125, 125);
    }
}

void MCUButton::read()
{
    _bounce.update();
    if (_bounce.fell()) {
        // MCU Note Bang: NoteOn immediately followed by NoteOff — no local LED toggle
        usbMIDI.sendNoteOn(_noteNum, 127, 1);   // channel 1, velocity 127 = pressed
        usbMIDI.sendNoteOff(_noteNum, 0, 1);    // channel 1, velocity 0 = released
        // DO NOT update LED here — Logic Pro sends LED state via NoteOn/Off callbacks
    }
}

void MCUButton::setLedState(uint8_t velocity)
{
    if (!_hasLed) return;
    if (velocity == 0) {
        SoftPWMSet(_ledPin, 0);                 // off
    } else if (velocity == 127) {
        SoftPWMSet(_ledPin, LED_MAX_BRIGHTNESS); // capped brightness — stays within USB 500mA budget
    }
    // velocity == 1 (blink) is handled in Phase 2 — ignore for now
}

int MCUButton::getLedPin() const { return _ledPin; }
int MCUButton::getNoteNum() const { return _noteNum; }
```

**Key invariants:**
- `read()` sends Note Bang only — never touches LED
- `setLedState()` is the only path that changes LED brightness
- `_hasLed = (ledPin >= 0)`: buttons with ledPin = -1 skip all SoftPWM calls (Stop and Play have no LEDs per pinDefines.h — check actual LED pin values; if those buttons do have pin assignments, keep _hasLed = true but note that CONTEXT.md says Stop/Play have no physical LED hardware)

```cpp
#include <Arduino.h>
#include <map>

class NoteRegistry
{
public:
    void registerButton(uint8_t noteNum, MCUButton* btn);
    MCUButton* getButton(uint8_t noteNum) const;

    std::map<uint8_t, MCUButton*> noteToButton;
};

#endif
```

**src/noteRegistry.cpp:**
```cpp
#include "noteRegistry.h"

void NoteRegistry::registerButton(uint8_t noteNum, MCUButton* btn)
{
    noteToButton[noteNum] = btn;
}

MCUButton* NoteRegistry::getButton(uint8_t noteNum) const
{
    auto it = noteToButton.find(noteNum);
    if (it != noteToButton.end()) {
        return it->second;
    }
    return nullptr;
}
```
